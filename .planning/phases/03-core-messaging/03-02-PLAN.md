8. **Send handler:**
```javascript
async function onSend() {
  const text   = textEl.value
  const key    = keyEl.value
  const cipher = cipherEl.value
  errEl.textContent = ''

  if (!text.trim()) { errEl.textContent = ''; return }   // silent no-op on empty text per CHAT-12
  if (!key)         { errEl.textContent = 'KEY REQUIRED'; return }

  // Persist prefs (SEC-05 — never sent to server)
  putSettings(convId, { cipher, key })

  // Apply historical cipher (encode=true), then AES-encrypt the cipher-text
  const cipherText = applyCipher(text, cipher, key, true)
  let payload, iv, salt
  try {
    ({ payload, iv, salt } = await encrypt(cipherText, key))
  } catch (_e) {
    errEl.textContent = 'SEND FAILED — TRY AGAIN'
    return
  }

  // Optimistic insert
  const tempId = 'opt-' + Date.now() + '-' + Math.random().toString(36).slice(2, 8)
  const optimistic = {
    id: tempId,
    conversation_id: convId,
    sender_id: user.id,
    payload, iv, salt,
    created_at: new Date().toISOString()
  }
  messages.push(optimistic)
  seenIds.add(tempId)
  decoded[tempId] = text   // we know our own plaintext
  textEl.value = ''
  renderMessages()

  // Transport call
  const { result: row, error } = await transport.sendMessage(convId, user.id, payload, iv, salt)
  if (error || !row) {
    // Rollback
    messages = messages.filter(m => m.id !== tempId)
    seenIds.delete(tempId)
    delete decoded[tempId]
    renderMessages()
    errEl.textContent = 'SEND FAILED — TRY AGAIN'
    return
  }
  // Promote optimistic id to server id; preserve plaintext mapping
  const idx = messages.findIndex(m => m.id === tempId)
  if (idx !== -1) {
    messages[idx] = row
    seenIds.delete(tempId)
    seenIds.add(row.id)
    decoded[row.id] = text
    delete decoded[tempId]
    renderMessages()
  }
}
```

9. **Realtime subscription (D-06, D-07):**
```javascript
const unsubMessages = transport.subscribeMessages(convId, (msg) => {
  if (seenIds.has(msg.id)) return  // dedupe optimistic echo
  seenIds.add(msg.id)
  messages.push(msg)
  renderMessages()
})
```
